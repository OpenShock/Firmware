cmake_minimum_required(VERSION 3.16.0)

# --- OpenShock generated build headers ---------------------------------------
# gen_env_header.py writes openshock_board.h and openshock_version.h into ${CMAKE_BINARY_DIR}/config.
# Neither is force-included: the commit in every translation unit would defeat ccache.
# OPENSHOCK_BOARD (-D, from scripts/build.py) is empty for a variant build, which package_boards.py stamps later.
set(OPENSHOCK_BOARD "" CACHE STRING "OpenShock board to stamp (boards/<name>.defaults); empty for a variant build")

# --- ESP-IDF without an exported shell ------------------------------------------
# `cmake --preset <board>` (CLion, a plain terminal) has no IDF_PATH. scripts/utils/idfenv.py finds an
# install registered by the Espressif installer and prints its environment; apply it to this configure.
# Compiler and tool paths end up absolute in the cache, so the build needs no activation either.
if(NOT DEFINED ENV{IDF_PATH})
  find_program(_os_host_python NAMES python3 python REQUIRED)
  execute_process(
    COMMAND ${_os_host_python} ${CMAKE_CURRENT_LIST_DIR}/scripts/utils/idfenv.py --dump
    OUTPUT_VARIABLE _os_idf_env RESULT_VARIABLE _os_idf_rc OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT _os_idf_rc EQUAL 0)
    message(FATAL_ERROR "IDF_PATH is not set and no ESP-IDF install was found: run the ESP-IDF export script first")
  endif()
  # Values hold ';' (PATH on Windows), so hide it from CMake's list splitting until each line is cut up.
  string(REPLACE ";" "@@SEMI@@" _os_idf_env "${_os_idf_env}")
  string(REPLACE "\n" ";" _os_idf_env "${_os_idf_env}")
  foreach(_os_line IN LISTS _os_idf_env)
    string(REGEX MATCH "^([^=]+)=(.*)$" _os_m "${_os_line}")
    if(_os_m)
      set(_os_key "${CMAKE_MATCH_1}")
      string(REPLACE "@@SEMI@@" ";" _os_value "${CMAKE_MATCH_2}")
      if(_os_key STREQUAL "PYTHON")
        if(NOT DEFINED PYTHON)
          set(PYTHON "${_os_value}")
        endif()
      else()
        set(ENV{${_os_key}} "${_os_value}")
      endif()
    endif()
  endforeach()
  message(STATUS "ESP-IDF activated from $ENV{IDF_PATH}")
  set(_os_idf_activated TRUE)
endif()

# --- Board from the command line --------------------------------------------------
# scripts/build.py passes IDF_TARGET, SDKCONFIG_DEFAULTS and SDKCONFIG itself. A preset passes only
# OPENSHOCK_BOARD; derive the rest from boards/<board>.defaults the way scripts/utils/boards.py does:
# keep the Kconfig lines (CONFIG_* and `# CONFIG_* is not set`), drop the stamped OPENSHOCK_* pins.
if(OPENSHOCK_BOARD AND NOT DEFINED SDKCONFIG_DEFAULTS)
  set(_os_board_file ${CMAKE_CURRENT_LIST_DIR}/boards/${OPENSHOCK_BOARD}.defaults)
  if(NOT EXISTS ${_os_board_file})
    message(FATAL_ERROR "Unknown board '${OPENSHOCK_BOARD}': no ${_os_board_file}")
  endif()
  file(STRINGS ${_os_board_file} _os_board_lines)
  set(_os_board_kconfig "# Board ${OPENSHOCK_BOARD}, without its pins (generated by CMakeLists.txt)\n")
  foreach(_os_line IN LISTS _os_board_lines)
    string(STRIP "${_os_line}" _os_line)
    if(_os_line MATCHES "^CONFIG_" OR _os_line MATCHES "^# CONFIG_.* is not set$")
      string(APPEND _os_board_kconfig "${_os_line}\n")
    endif()
    if(NOT DEFINED IDF_TARGET AND _os_line MATCHES "^CONFIG_IDF_TARGET=\"([^\"]+)\"")
      set(IDF_TARGET ${CMAKE_MATCH_1})
    endif()
  endforeach()
  # The partition table follows from the flash size, as in scripts/build.py: sdkconfig.defaults names the 4 MB
  # table, a board with 8 MB or more gets partitions/ota_8mb.csv.
  string(REGEX MATCH "CONFIG_ESPTOOLPY_FLASHSIZE_([0-9]+)MB=y" _os_m "${_os_board_kconfig}")
  if(_os_m AND CMAKE_MATCH_1 GREATER_EQUAL 8)
    string(APPEND _os_board_kconfig "CONFIG_PARTITION_TABLE_CUSTOM=y\nCONFIG_PARTITION_TABLE_CUSTOM_FILENAME=\"partitions/ota_8mb.csv\"\n")
  endif()
  set(_os_board_fragment ${CMAKE_BINARY_DIR}/openshock_board.defaults)
  file(WRITE ${_os_board_fragment} "${_os_board_kconfig}")
  set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${_os_board_fragment}")
  if(NOT DEFINED SDKCONFIG)
    set(SDKCONFIG ${CMAKE_BINARY_DIR}/sdkconfig)
  endif()
endif()

# --- Interpreter for the project's scripts --------------------------------------------------------
# gen_env_header.py, stamp_board.py and gen_staticfs.py need requirements.txt, which must not go into
# ESP-IDF's venv (its constraints conflict). scripts/build.py passes its own interpreter as
# OPENSHOCK_PYTHON; a preset build (CLion, plain cmake) takes .venv/ in the repository. Without either,
# the interpreter idf.py passes as PYTHON (ESP-IDF's venv) or the one on PATH is tried and checked.
set(OPENSHOCK_PYTHON "" CACHE FILEPATH "Interpreter with requirements.txt installed, for the project's scripts")
if(OPENSHOCK_PYTHON)
  set(_os_python ${OPENSHOCK_PYTHON})
elseif(EXISTS ${CMAKE_CURRENT_LIST_DIR}/.venv/Scripts/python.exe)
  set(_os_python ${CMAKE_CURRENT_LIST_DIR}/.venv/Scripts/python.exe)
elseif(EXISTS ${CMAKE_CURRENT_LIST_DIR}/.venv/bin/python)
  set(_os_python ${CMAKE_CURRENT_LIST_DIR}/.venv/bin/python)
elseif(DEFINED PYTHON)
  set(_os_python ${PYTHON})
else()
  set(_os_python python)
endif()

# Every pin of requirements.txt, checked against that interpreter now: a missing package fails the
# configure with the list instead of the build halfway through. Nothing is installed here.
execute_process(
  COMMAND ${_os_python} ${CMAKE_CURRENT_LIST_DIR}/scripts/utils/pyreqs.py
          --check ${CMAKE_CURRENT_LIST_DIR}/requirements.txt
  RESULT_VARIABLE _os_req_rc ERROR_VARIABLE _os_req_err OUTPUT_QUIET)
if(NOT _os_req_rc EQUAL 0)
  string(STRIP "${_os_req_err}" _os_req_err)
  message(FATAL_ERROR "${_os_req_err}")
endif()
message(STATUS "Project scripts run with ${_os_python}")

# scripts/build.py passes -DIDF_TARGET; with idf.py it comes from `idf.py set-target <chip>` or the
# environment. Guessing instead (ESP-IDF's own fallback is esp32) could contradict the sdkconfig this
# build layers, and the chip label in the generated header would then disagree with the image.
if(DEFINED IDF_TARGET)
  set(_os_target ${IDF_TARGET})
elseif(DEFINED ENV{IDF_TARGET})
  set(_os_target $ENV{IDF_TARGET})
else()
  message(FATAL_ERROR "IDF_TARGET is not set: pass -DIDF_TARGET=<chip>, export IDF_TARGET, or build through scripts/build.py")
endif()

set(OPENSHOCK_CONFIG_DIR ${CMAKE_BINARY_DIR}/config)
set(OPENSHOCK_VERSION_HEADER ${OPENSHOCK_CONFIG_DIR}/openshock_version.h)
set(_os_gen_command
    ${_os_python} ${CMAKE_CURRENT_LIST_DIR}/scripts/gen_env_header.py
    --out-dir ${OPENSHOCK_CONFIG_DIR}
    --target ${_os_target})

execute_process(COMMAND ${_os_gen_command} RESULT_VARIABLE _os_env_rc)
if(NOT _os_env_rc EQUAL 0)
  message(FATAL_ERROR "gen_env_header.py failed (rc=${_os_env_rc})")
endif()

# The bootloader is a nested CMake configure that ninja launches with the caller's environment, so
# a self-activated build hands it ninja and the compiler directory explicitly; the top level resolved
# both through the activated PATH and caches them as absolute paths.
if(_os_idf_activated)
  find_program(_os_ninja ninja REQUIRED)
  set(CMAKE_MAKE_PROGRAM ${_os_ninja} CACHE FILEPATH "ninja from the activated ESP-IDF tools")
  file(STRINGS $ENV{IDF_PATH}/tools/cmake/toolchain-${_os_target}.cmake _os_prefix_line REGEX "_CMAKE_TOOLCHAIN_PREFIX")
  string(REGEX MATCH "_CMAKE_TOOLCHAIN_PREFIX ([^ )]+)" _os_m "${_os_prefix_line}")
  find_program(_os_gcc ${CMAKE_MATCH_1}gcc REQUIRED)
  get_filename_component(_os_gcc_dir ${_os_gcc} DIRECTORY)
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

if(_os_idf_activated)
  idf_build_set_property(EXTRA_CMAKE_ARGS "-DCMAKE_MAKE_PROGRAM=${_os_ninja}" APPEND)
  idf_build_set_property(EXTRA_CMAKE_ARGS "-DCMAKE_PROGRAM_PATH=${_os_gcc_dir}" APPEND)
endif()

project(firmware)

# --- ccache by absolute path ---------------------------------------------------
# idf.py configures with CCACHE_ENABLE, and ESP-IDF then launches compiles through a bare `ccache`,
# which only resolves inside an exported shell. CLion or a plain `cmake --build` on the same tree
# fails every compile with "CreateProcess failed". ESP-IDF already found the program; use that path.
if(CCACHE_ENABLE AND CCACHE_FOUND)
  set_property(GLOBAL PROPERTY RULE_LAUNCH_COMPILE "${CCACHE_FOUND}")
endif()

# The version comes from git and the environment, which CMake cannot track, so refresh the header on
# every build. It is written only when it changes, so an unchanged version rebuilds nothing.
add_custom_target(openshock_version_header ALL
  BYPRODUCTS ${OPENSHOCK_VERSION_HEADER}
  COMMAND ${_os_gen_command} --version-only
  COMMENT "Refreshing openshock_version.h"
  VERBATIM)
# Order the refresh ahead of Version.cpp, the only translation unit that includes it.
idf_component_get_property(_os_common_lib common COMPONENT_LIB)
add_dependencies(${_os_common_lib} openshock_version_header)

# --- Board descriptor stamp -------------------------------------------------
# stamp_board.py stamps the board into the .bin as soon as it is written, so `idf.py flash` carries it.
if(OPENSHOCK_BOARD)
  idf_build_get_property(_os_build_dir BUILD_DIR)
  add_custom_command(TARGET gen_project_binary POST_BUILD
    COMMAND ${_os_python} ${CMAKE_CURRENT_LIST_DIR}/scripts/stamp_board.py stamp
            --image ${_os_build_dir}/${CMAKE_PROJECT_NAME}.bin
            --board ${OPENSHOCK_BOARD}
    COMMENT "Stamping board descriptor for ${OPENSHOCK_BOARD} into ${CMAKE_PROJECT_NAME}.bin"
    VERBATIM)
endif()

# --- Compile database for clangd --------------------------------------------
# .clangd reads build/compile_commands.json. scripts/build.py copies it there after its builds; a
# `cmake --build --preset <board>` has no script, so publish it from the build itself.
set(_os_compile_db_dir ${CMAKE_CURRENT_LIST_DIR}/build)
if(NOT CMAKE_BINARY_DIR STREQUAL _os_compile_db_dir)
  add_custom_command(TARGET gen_project_binary POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory ${_os_compile_db_dir}
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            ${CMAKE_BINARY_DIR}/compile_commands.json ${_os_compile_db_dir}/compile_commands.json
    COMMENT "Publishing compile_commands.json for clangd"
    VERBATIM)
endif()

# --- Captive-portal asset image (www) ----------------------------------------
# gen_staticfs.py packs the frontend into the flat `www` image; skipped when the partition table has no such partition.
partition_table_get_partition_info(_www_offset "--partition-name www" "offset")
partition_table_get_partition_info(_www_size   "--partition-name www" "size")

if("${_www_offset}" STREQUAL "" OR "${_www_size}" STREQUAL "")
  message(WARNING "No `www` partition in the active partition table; "
                  "skipping asset image generation.")
else()
  set(_www_image ${CMAKE_BINARY_DIR}/www.bin)

  # CI never builds this target: its staticfs job runs gen_staticfs.py --skip-frontend-build once for every variant.
  add_custom_target(www ALL
    BYPRODUCTS ${_www_image}
    COMMAND ${_os_python} ${CMAKE_CURRENT_LIST_DIR}/scripts/gen_staticfs.py
            --frontend-dir ${CMAKE_CURRENT_LIST_DIR}/frontend
            --output       ${_www_image}
            --max-size     ${_www_size}
    COMMENT "Building captive-portal asset image (www.bin) for www"
    VERBATIM USES_TERMINAL)

  # Register the image with the flasher so `idf.py flash` writes it to the
  # www offset alongside the app and partition table.
  esptool_py_flash_target_image(flash "www" "${_www_offset}" "${_www_image}")
endif()
